    <script>
        // CONFIGURAZIONE SUPABASE (Sostituisci qui i tuoi dati reali presi dalla dashboard)

        const supabase = supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

        const myUsername = 'me';

        // REGISTRAZIONE SERVICE WORKER (PWA)
        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('./sw.js')
                .then(() => console.log('PWA Service Worker attivo'))
                .catch(err => console.log('Errore Service Worker:', err));

        // CARICA POST
        async function fetchPosts() {
            const { data, error } = await supabase.from('posts').select('*').order('created_at', { ascending: false });
            if (error) return console.error('Errore post:', error);
            
            const container = document.getElementById('posts-list');
            if (!data || data.length === 0) {
                container.innerHTML = '<p style="text-align: center; color: var(--text-secondary);">Nessun post presente.</p>';
                return;
            }
            container.innerHTML = data.map(post => `
                <article class="post-card">
                    <img src="${post.image}" class="post-image" alt="Post">
                    <div class="post-body">
                        <strong>@${post.author}</strong>
                        <p style="margin-top: 6px; color: #cbd5e1;">${post.caption}</p>
                    </div>
                </article>
            `).join('');
        }

        // CARICA REELS
        async function fetchReels() {
            const { data, error } = await supabase.from('reels').select('*').order('created_at', { ascending: false });
            if (error) return console.error('Errore reels:', error);

            const container = document.getElementById('reels-list');
            if (!data || data.length === 0) {
                container.innerHTML = '<p style="text-align: center; color: var(--text-secondary);">Nessun video presente.</p>';
                return;
            }
            container.innerHTML = data.map(reel => `
                <div class="reel-card">
                    <video class="reel-video" loop autoplay muted playsinline src="${reel.video_url}"></video>
                    <div class="reel-overlay">
                        <div>
                            <strong>@${reel.author}</strong>
                            <p style="margin-top: 6px; color: white;">${reel.caption}</p>
                        </div>
                    </div>
                </div>
            `).join('');
        }

        // CARICA MESSAGGI CHAT
        async function fetchMessages() {
            const { data, error } = await supabase.from('messages').select('*').order('created_at', { ascending: true });
            if (error) return console.error('Errore messaggi:', error);

            renderMessages(data);
        }

        function renderMessages(messages) {
            const container = document.getElementById('chat-messages');
            if (!messages || messages.length === 0) {
                container.innerHTML = '<p style="text-align: center; color: var(--text-secondary);">Inizia la conversazione!</p>';
                return;
            }
            container.innerHTML = messages.map(msg => {
                const isMe = msg.sender === myUsername;
                return `
                    <div class="message-bubble ${isMe ? 'me' : 'other'}">
                        <div style="font-size: 11px; opacity: 0.7; margin-bottom: 2px;">@${msg.sender}</div>
                        <div>${msg.text}</div>
                    </div>
                `;
            }).join('');
            container.scrollTop = container.scrollHeight;
        }

        // INVIA MESSAGGIO
        async function sendMessage() {
            const input = document.getElementById('chat-input');
            const text = input.value.trim();
            if (!text) return;

            input.value = '';
            
            const { error } = await supabase.from('messages').insert([{
                sender: myUsername,
                text: text
            }]);

            if (error) console.error('Errore invio messaggio:', error);
        }

        function handleKeyPress(e) {
            if (e.key === 'Enter') sendMessage();
        }

        // ISCRIVITI AI MESSAGGI IN TEMPO REALE
        function subscribeRealtimeChat() {
            supabase
                .channel('public:messages')
                .on('postgres_changes', { event: 'INSERT', schema: 'public', table: 'messages' }, payload => {
                    fetchMessages();
                })
                .subscribe();
        }

        // UPLOAD MEDIA
        async function uploadToDatabase() {
            const type = document.getElementById('create-type').value;
            const url = document.getElementById('create-media-url').value;
            const caption = document.getElementById('create-caption').value;

            if (!url) {
                alert('Inserisci un URL valido');
                return;
            }

            if (type === 'post') {
                await supabase.from('posts').insert([{
                    author: 'utente_app',
                    avatar: 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=150',
                    image: url,
                    caption: caption
                }]);
                fetchPosts();
                switchView('posts');
            } else {
                await supabase.from('reels').insert([{
                    author: 'utente_app',
                    avatar: 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?w=150',
                    video_url: url,
                    caption: caption,
                    audio: 'Suono originale'
                }]);
                fetchReels();
                switchView('reels');
            }
            closeCreateModal();
        }

        function switchView(view) {
            document.querySelectorAll('.view-section').forEach(el => el.classList.remove('active'));
            document.querySelectorAll('.bottom-nav-item, .nav-item').forEach(el => el.classList.remove('active'));
            
            document.getElementById(`view-${view}`).classList.add('active');
            
            const navEl = document.getElementById(`nav-${view}`);
            const bnavEl = document.getElementById(`bnav-${view}`);
            if (navEl) navEl.classList.add('active');
            if (bnavEl) bnavEl.classList.add('active');
        }

        function openCreateModal() {
            document.getElementById('modal-create').classList.add('active');
        }

        function closeCreateModal() {
            document.getElementById('modal-create').classList.remove('active');
        }

        // INIT
        window.addEventListener('DOMContentLoaded', () => {
            fetchPosts();
            fetchReels();
            fetchMessages();
            subscribeRealtimeChat();
        });
